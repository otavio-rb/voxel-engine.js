@voxel/engine

Motor voxel 3D para navegadores, em TypeScript e Three.js, com a simulação e a geração de mundo isoladas em Web Workers dedicados — a main thread só possui o canvas e o input.A 3D voxel engine for the browser, in TypeScript and Three.js, with simulation and world generation isolated in dedicated Web Workers — the main thread only owns the canvas and input.

Quick StartQuick Start

src/main.ts
import { VoxelEngine } from '@voxel/engine';

const canvas = document.querySelector('canvas')!;
const worker = new Worker(new URL('./workers/engine.worker.ts', import.meta.url), { type: 'module' });

const engine = new VoxelEngine({ canvas, worker });
engine.on('stats', (s) => console.log(s.fps, s.triangles));
engine.start();
src/workers/engine.worker.ts
import { runEngineWorker, type EnginePlugin } from '@voxel/engine/worker';

const hydrationPlugin: EnginePlugin = {
  name: 'hydration',
  setup(ctx) {
    ctx.on('fixedUpdate', ({ deltaMs }) => { /* 1 vez por tick */ });
  }
};

runEngineWorker({
  createChunkWorker: () => new Worker(new URL('./chunk.worker.ts', import.meta.url), { type: 'module' }),
  targetTps: 60,
  plugins: [hydrationPlugin]
});
src/workers/chunk.worker.ts
import { runChunkWorker, type WorldGenerator } from '@voxel/engine/chunk-worker';
